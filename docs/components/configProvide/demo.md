---
title: McConfigProvider 全局化配置
desc: 为组件提供统一的全局化配置。
---

按需引入路径：

```ts
import { McConfigProvider } from '@matechat/core';
```

`McConfigProvider` 使用 Vue 的 [provide / inject](https://vuejs.org/v2/api/#provide-inject) 特性，只需在应用外围包裹一次即可全局生效。

### 国际化配置

:::demo

```vue
<template>
  <button class="demo-btn" @click="toggleLanguage">
    {{ language === 'zh-cn' ? '切换为英文' : '切换为中文' }}
  </button>
  <McConfigProvider :locale="language" :customLocaleMessages="customLocale">
    <McInput
      ref="inputEl"
      :value="inputValue"
      :maxLength="2000"
      :loading="loading"
      :autoClear="false"
      showCount
      @change="onInputChange"
      @submit="onSubmit"
      @cancel="onCancel"
    />
    <div class="demo-tips">{{ mcI18n.t('aiTips') }}</div>
  </McConfigProvider>
</template>

<script setup lang="ts">
import { ref } from 'vue';
import { McInput } from '@matechat/core';
import { useMcI18n } from '@matechat/core/locale';

const inputEl = ref();
const inputValue = ref('');
const loading = ref(false);
const language = ref('zh-cn'); // 'zh-cn' | 'en-us'
const mcI18n = useMcI18n();
const customLocale = ref({
  'zh-cn': {
    Input: {
      send: '自定义发送',
    },
    aiTips: '内容由AI生成，可能不准确，请注意核实',
  },
  'en-us': {
    Input: {
      send: 'Custom send',
    },
    aiTips: 'Content generated by AI, may not be accurate, please verify it.',
  },
});

const onInputChange = (e) => {
  console.log('input change---', e);
};
const onSubmit = (e) => {
  loading.value = true;
  console.log('input submit---', e);
  setTimeout(() => {
    loading.value = false;
    inputValue.value = '';
    inputEl.value.clearInput();
  }, 5000);
};
const onCancel = () => {
  loading.value = false;
  console.log('input cancel');
};
const toggleLanguage = () => {
  language.value = language.value === 'zh-cn' ? 'en-us' : 'zh-cn';
};
</script>
<style scoped>
.demo-btn {
  background-color: var(--devui-primary, #5e7ce0);
  border-color: transparent;
  color: var(--devui-light-text, #ffffff);
  min-width: 64px;
  border-radius: 4px;
  cursor: pointer;
  position: relative;
  margin-bottom: 16px;
  padding: var(--devui-btn-padding, 0 20px);
  font-size: var(--devui-font-size-md, 12px);
  height: 32px;
  line-height: var(--devui-line-height-base, 1.5);
  border-width: 1px;
  border-color: transparent;

  &:active {
    background-color: var(--devui-primary-active, #344899);
  }
}
.demo-tips {
  font-size: 10px;
  color: var(--devui-text-2, #999999);
  margin-top: 8px;
  text-align: center;
}
</style>
```

### 主题配置

:::demo

```vue
<template>
  <div class="demo-color-picker">
    <span>选择主题颜色：</span>
    <input type="color" :value="themeColor" @input="onColorChange" />
  </div>
  <McConfigProvider :theme="theme">
    <button class="demo-btn" @click="onResetTheme">恢复</button>
    <SubComponent style="margin-bottom: 16px;" />
    <McInput :value="inputValue" :maxLength="2000" />
  </McConfigProvider>
</template>

<script setup lang="ts">
import { ref, reactive, computed, h, defineComponent } from 'vue';
import { ThemeServiceInit, infinityTheme } from 'devui-theme';
import { McInput, useMcConfigInject } from '@matechat/core';

const inputValue = ref('');
const theme = ref('infinityTheme');
// 详细说明可参考：https://matechat.gitcode.com/use-guide/cli-theme.html
const customThemeConfig = {
  id: 'custom',
  name: '自定义主题',
  autoDeduction: true,
  data: { 'devui-brand': '#F7A2AD' }, // 品牌主色
};

const themeColor = computed(() => {
  if (theme.value?.data?.['devui-brand']) {
    return theme.value.data['devui-brand'];
  }
  return infinityTheme.data['devui-brand'];
});

const SubComponent = defineComponent({
  name: 'SubComponent',
  setup() {
    const mcConfig = useMcConfigInject();
    const tips = computed(() => {
      if (typeof mcConfig.theme === 'string') {
        return `当前主题：${mcConfig.theme}`;
      } else if (mcConfig.theme.autoDeduction) {
        return `当前自定义主题基色：${mcConfig.theme.data?.['devui-brand']}`;
      } else {
        return `当前主题：${mcConfig.theme.id}`;
      }
    });
    return () => h('div', {}, [tips.value]);
  },
});

// 初始是 infinityTheme 无限主题
const themeService = ThemeServiceInit({
  infinityTheme,
}, 'infinityTheme');

const onColorChange = (evt: Event) => {
  const customColor = evt.target.value;
  const themeConfig = {
    ...customThemeConfig,
    id: `custom-${customColor}`,
    data: {
      'devui-brand': customColor, // 品牌主色
    },
  }
  theme.value = themeConfig;
};
const onResetTheme = () => {
  theme.value = 'infinityTheme';
};
</script>
<style scoped>
.demo-btn {

  border-color: transparent;
  color: var(--devui-light-text, #ffffff);

  position: relative;
  margin-bottom: 16px;
  padding: var(--devui-btn-padding, 0 20px);
  font-size: var(--devui-font-size-md, 12px);
  height: 32px;
  line-height: var(--devui-line-height-base, 1.5);
  border-width: 1px;
  border-color: transparent;

  &:active {
    background-color: var(--devui-primary-active, #344899);
  }
}
.demo-tips {
  font-size: 10px;
  color: var(--devui-text-2, #999999);
  margin-top: 8px;
  text-align: center;
}
.demo-color-picker {
  display: flex;

  input {
    width: 24px;
    height: 24px;
  }
}
</style>
```

### 服务化方式调用

注意：如果想要让配置全局生效，需要在根组件中调用 `useConfigProvider` 服务函数。

```ts
import { useConfigProvider } from '@matechat/core';
import type { ConfigProviderProps } from '@matechat/core/ConfigProvider/types';

const customConfig = reactive<ConfigProviderProps>({
  locale: 'zh-cn', // 'zh-cn' | 'en-us'
  customLocaleMessages: {
    'zh-cn': {
      Input: {
        send: '自定义发送',
      },
      aiTips: '内容由AI生成，可能不准确，请注意核实',
    },
    'en-us': {
      Input: {
        send: 'Custom send',
      },
      aiTips: 'Content generated by AI, may not be accurate, please verify it.',
    },
  },
});
useConfigProvider(customConfig);
```
