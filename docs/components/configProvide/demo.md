---
title: McConfigProvider 全局化配置
desc: 为组件提供统一的全局化配置。
---

按需引入路径：

```ts
import { McConfigProvider } from '@matechat/core';
```

`McConfigProvider` 使用 Vue 的 [provide / inject](https://vuejs.org/v2/api/#provide-inject) 特性，只需在应用外围包裹一次即可全局生效。

### 国际化配置

:::demo

```vue
<template>
  <button class="demo-btn" @click="toggleLanguage">
    {{ language === 'zh-cn' ? '切换为英文' : '切换为中文' }}
  </button>
  <McConfigProvider :locale="language" :customLocaleMessages="customLocale">
    <McInput
      ref="inputEl"
      :value="inputValue"
      :maxLength="2000"
      :loading="loading"
      :autoClear="false"
      showCount
      @change="onInputChange"
      @submit="onSubmit"
      @cancel="onCancel"
    />
    <div class="demo-tips">{{ mcI18n.t('aiTips') }}</div>
  </McConfigProvider>
</template>

<script setup lang="ts">
import { ref } from 'vue';
import { McInput } from '@matechat/core';
import { useMcI18n } from '@matechat/core/locale';

const inputEl = ref();
const inputValue = ref('');
const loading = ref(false);
const language = ref('zh-cn'); // 'zh-cn' | 'en-us'
const mcI18n = useMcI18n();
const customLocale = ref({
  'zh-cn': {
    Input: {
      send: '自定义发送',
    },
    aiTips: '内容由AI生成，可能不准确，请注意核实',
  },
  'en-us': {
    Input: {
      send: 'Custom send',
    },
    aiTips: 'Content generated by AI, may not be accurate, please verify it.',
  },
});

const onInputChange = (e) => {
  console.log('input change---', e);
};
const onSubmit = (e) => {
  loading.value = true;
  console.log('input submit---', e);
  setTimeout(() => {
    loading.value = false;
    inputValue.value = '';
    inputEl.value.clearInput();
  }, 5000);
};
const onCancel = () => {
  loading.value = false;
  console.log('input cancel');
};
const toggleLanguage = () => {
  language.value = language.value === 'zh-cn' ? 'en-us' : 'zh-cn';
};
</script>
<style scoped>
.demo-btn {
  background-color: var(--devui-primary, #5e7ce0);
  border-color: transparent;
  color: var(--devui-light-text, #ffffff);
  min-width: 64px;
  border-radius: var(--devui-border-radius, 2px);
  cursor: pointer;
  position: relative;
  margin-bottom: 16px;
  padding: var(--devui-btn-padding, 0 20px);
  font-size: var(--devui-font-size-md, 12px);
  height: 32px;
  line-height: var(--devui-line-height-base, 1.5);
  border-width: 1px;
  border-color: transparent;

  &:active {
    background-color: var(--devui-primary-active, #344899);
  }
}
.demo-tips {
  font-size: 10px;
  color: var(--devui-text-2, #999999);
  margin-top: 8px;
  text-align: center;
}
</style>
```

### 服务化方式调用

注意：如果想要让配置全局生效，需要在根组件中调用 `useConfigProvider` 服务函数。

:::demo

```vue
<template>
  <button class="demo-btn" @click="toggleLanguage">
    {{ customConfig.locale === 'zh-cn' ? '切换为英文' : '切换为中文' }}
  </button>
  <McInput
    ref="inputEl"
    :value="inputValue"
    :maxLength="2000"
    :loading="loading"
    :autoClear="false"
    showCount
    @change="onInputChange"
    @submit="onSubmit"
    @cancel="onCancel"
  />
  <div class="demo-tips">{{ mcI18n.t('aiTips') }}</div>
</template>

<script setup lang="ts">
import { reactive, ref } from 'vue';
import { useConfigProvider, McInput } from '@matechat/core';
import { useMcI18n } from '@matechat/core/locale';

const inputEl = ref();
const inputValue = ref('');
const loading = ref(false);
const mcI18n = useMcI18n();
const customConfig = reactive({
  locale: 'zh-cn', // 'zh-cn' | 'en-us'
  customLocaleMessages: {
    'zh-cn': {
      Input: {
        send: '自定义发送',
      },
      aiTips: '内容由AI生成，可能不准确，请注意核实',
    },
    'en-us': {
      Input: {
        send: 'Custom send',
      },
      aiTips: 'Content generated by AI, may not be accurate, please verify it.',
    },
  },
})
useConfigProvider(customConfig, 'demo');

const onInputChange = (e) => {
  console.log('input change---', e);
};
const onSubmit = (e) => {
  loading.value = true;
  console.log('input submit---', e);
  setTimeout(() => {
    loading.value = false;
    inputValue.value = '';
    inputEl.value.clearInput();
  }, 5000);
};
const onCancel = () => {
  loading.value = false;
  console.log('input cancel');
};
const toggleLanguage = () => {
  customConfig.locale = customConfig.locale === 'zh-cn' ? 'en-us' : 'zh-cn';
};
</script>
```
